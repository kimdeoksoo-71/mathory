export function trimLeadingBlankLines(t: string, type?: string): string {
  if (type && CASE_TYPES.has(type) && /^[ \t]*\n/.test(t)) {            // 이어짓기 — 빈 첫 줄 하나는 의미다
    const rest = t.replace(/^\s*\n/, '');
    return rest ? '\n' + rest : '';
  }
  return t.replace(/^\s*\n/, '');
}
export const trimTrailingBlankLines = (t: string) => t.replace(/\n\s*$/, '');
export const trimBlockEdges = (t: string, type?: string) => trimTrailingBlankLines(trimLeadingBlankLines(t, type));

// MarkdownEditor 핸들
trimEdges(type?: string): boolean {
  const view = viewRef.current; if (!view || view.hasFocus) return false;
  flushMathAscii(view);
  const s = view.state.doc.toString(); const next = trimBlockEdges(s, type);
  if (next === s) return false;
  view.dispatch({ changes: edgeChanges(s, type), effects: setSlots.of(null) });   // [v5] 앞·뒤 두 구간 — 통째 교체 금지
  return true;
}

// [v5] lib/blockEdges.ts — 위 trimBlockEdges와 결과가 같은 최소 changes(node로 표본 27조합 일치 확인 — 이어짓기·전부 공백·빈 문서 포함)
export function edgeChanges(s: string, type?: string): Array<{ from: number; to: number; insert: string }> {
  const rest = s.replace(/^\s*\n/, '');
  const head = s.length - rest.length;
  const keep = type && CASE_TYPES.has(type) && /^[ \t]*\n/.test(s) && rest ? '\n' : '';
  if (rest === '') return s ? [{ from: 0, to: s.length, insert: '' }] : [];
  const out: Array<{ from: number; to: number; insert: string }> = [];
  if (head && !(keep && head === 1 && s[0] === '\n')) out.push({ from: 0, to: head, insert: keep });
  const tail = rest.match(/\n\s*$/)?.[0].length ?? 0;
  if (tail) out.push({ from: s.length - tail, to: s.length, insert: '' });
  return out;
}

### 8-3. CLAUDE.md에 남길 규약(초안)
- **문항 메타 저장은 `onSaved`로 AppShell에 알린다(M10 A)** — 목록은 구독이 아니다(`getDocs` + `loadData`). 편집창에서 나가는 모든 길이 저장 완료 뒤 `applySavedMeta`를 거친다(폴더 변경만 재조회). 언마운트 저장은 진행 중 저장을 기다린다. 새 뷰 전환 핸들러를 만들 때 `loadData`를 따로 부를 필요가 없다.
- **전역 "바쁨"의 원천은 `lib/activityStore.ts`(M10 B)** — 배선은 `runVerifyFlow`·`runBatchVerify` 둘뿐(호출부에 두지 말 것). 파비콘은 `FaviconController` 하나가 `link[rel=icon]` href를 소유한다(다른 곳에서 href를 만지지 말 것). SVG 애니메이션은 Chrome이 재생하지 않는다 — canvas 프레임. 백그라운드 탭은 1Hz(5분 뒤 정지)가 상한 — "안 움직인다" 보고는 이것부터. 점은 진한 빨강 `#7A1F14` + 흰 테(팔레트 안).
- **AI 완성은 없다(M10 C)** — `/api/ai-complete`·`getAIProvider`·`resolveAIProviderModel`·`AI_PROVIDER`·`AI_MODEL`·⌘J 전부 삭제. 되살리지 말 것. `insertPlainText`는 OCR·61c·InkInbox가 쓴다.
- **세션 자동 이름은 `nextSessionName`(M10 E)** — `'일괄 검증'`·`'댓글'`은 예약(`ensureBatchSession`이 이름으로 찾는다). 알약 활성 재클릭 = 이름 변경(`e.detail`, 타이머 금지). `+ 새 토론`은 오너만.
- **선택 감싸기 셋(`\left…\right`·`\mathrm{}`·툴바 rm/text)은 감싼 뒤 커서를 닫는 구분자 뒤에 둔다(M10 F·G)** — 선택 유지는 `**` 토글뿐. `\mathrm` 트리거는 **`"`**(수식 안 선택 + 큰따옴표) — 영문 글자(`r` 등)를 트리거로 쓰지 말 것: 한글 IME가 조합하는 키라 선택이 `ㄱ`으로 먼저 대체돼 inputHandler·keymap·keydown 어디서도 못 잡는다(M10 v1 §1-F3). 기호 키(`(`·`[`·`{`·`"`)는 조합 없이 ASCII로 오므로 D22 경로가 두 IME 모드에서 같다. 새 "선택 + 키" 기능은 반드시 기호 키로.
- **새 블록·타입 전환은 `focusBlockEnd`(M10 H)** — `focus()` 뒤 `setSelection(len,len)` · 50ms(ref 등록은 다음 커밋).
- **정돈 R0b = itemize·enumerate·description 벗기기(M10 I)** — R0 뒤·분할 앞·autoFix 무관. 결정적 세트(`autoFixDeterministicIssues`)에 넣지 말 것(OCR이 공유 · P-2가 제어열 불변을 고정).
- **블록 앞뒤 빈 줄의 정규식은 `trimBlockEdges` 하나(M10 J)** — 저장·정돈 R4·로드·분할·이탈·탭 전환이 같은 함수. 화면 반영은 **떠난 블록**의 CM에 두 구간 삭제 dispatch(통째 교체 금지 · `flushMathAscii` 먼저). 활성 블록은 건드리지 않는다. DOM blur는 이탈 신호가 아니다(68a 끊기·`commitComposition`이 같은 블록 안에서 blur를 낸다).
