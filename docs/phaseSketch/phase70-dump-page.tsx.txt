'use client';
// Phase 70a Stage 0 — 임시 덤프 페이지. app/dev70-dump/page.tsx로 복사해 쓰고 Stage 0 뒤 지운다(커밋하지 않는다).
// 로그인한 브라우저에서 http://localhost:3000/dev70-dump → [내려받기] → mathory-dump.json
// 본인 문항(휴지통 제외)의 제목·폴더 경로·탭·블록 원문만 담는다. 서버 전송 0 · 비밀값 0.
// ⚠ useAuth를 쓰지 않는다(구독마다 users/{uid} upsert) · AppShell 밖이라 단일 세션 claim도 없다(Phase 69 /ink와 같음).
import { useEffect, useState } from 'react';
import { onAuthStateChanged, type User } from 'firebase/auth';
import { auth } from '../../lib/firebase';
import { listProblems, listFolders, getProblemWithBlocks, TRASH_FOLDER_ID } from '../../lib/firestore';

export default function Dev70Dump() {
  const [user, setUser] = useState<User | null>(null);
  const [status, setStatus] = useState('로그인 확인 중…');

  useEffect(() => onAuthStateChanged(auth, (u) => {
    setUser(u);
    setStatus(u ? `${u.email} — 준비됨` : '로그인되어 있지 않습니다. 먼저 Mathory(/)에서 로그인하세요.');
  }), []);

  const run = async () => {
    if (!user) return;
    setStatus('문항 목록 읽는 중…');
    const [problems, folders] = await Promise.all([listProblems(user.uid), listFolders(user.uid)]);
    const byId = new Map(folders.map((f) => [f.id, f]));
    const pathOf = (fid?: string) => {
      const names: string[] = [];
      let cur = fid ? byId.get(fid) : undefined;
      while (cur) { names.unshift(cur.name); cur = cur.parent_id ? byId.get(cur.parent_id) : undefined; }
      return names.join('\\');
    };
    const live = problems.filter((p) => p.folder_id !== TRASH_FOLDER_ID);
    const out: unknown[] = [];
    for (let i = 0; i < live.length; i++) {
      setStatus(`블록 읽는 중… ${i + 1}/${live.length}`);
      const full = await getProblemWithBlocks(live[i].id);
      if (!full) continue;
      out.push({
        id: full.id, title: full.title, folder: pathOf(full.folder_id),
        tabs: (full.tabs || []).map((t) => ({ id: t.id, label: t.label })),
        tabBlocks: Object.fromEntries(Object.entries(full.tabBlocks).map(([k, bs]) => [k, bs.map((b) => ({
          id: b.id, type: b.type, raw_text: b.raw_text, block_key: (b as { block_key?: string }).block_key,
          imageWidth: b.imageWidth, imageTreatment: b.imageTreatment, imageGray: b.imageGray, svg_height: b.svg_height,
        }))])),
      });
    }
    const blob = new Blob([JSON.stringify({ exportedAt: new Date().toISOString(), problems: out }, null, 1)], { type: 'application/json' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'mathory-dump.json';
    a.click();
    setStatus(`완료 — ${out.length}문항`);
  };

  return (
    <div style={{ padding: 32, fontFamily: 'var(--font-ui)', fontSize: 15 }}>
      <h1 style={{ fontSize: 18, marginBottom: 12 }}>Phase 70 Stage 0 — 문항 덤프(임시)</h1>
      <p style={{ marginBottom: 16 }}>{status}</p>
      <button onClick={run} disabled={!user} style={{ padding: '8px 16px', fontSize: 14 }}>내려받기</button>
    </div>
  );
}
