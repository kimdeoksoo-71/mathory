import { useEffect, useState } from 'react';
import ExportDialog from '../../components/export/ExportDialog';
import type { Block, TabMeta } from '../../types/problem';

const R = String.raw;
const b = (id: string, type: string, raw_text: string, extra: Partial<Block> = {}) => ({ id, type, raw_text, order: 0, ...extra } as unknown as Block);

const tabs: TabMeta[] = [
  { id: 'question', label: '문제' } as TabMeta,
  { id: 'solution', label: '풀이' } as TabMeta,
];
const tabBlocks: Record<string, Block[]> = {
  question: [
    b('q1', 'text', R`함수 $f(x)=\frac{x^2+1}{x}$에 대하여 [4점]`),
    b('q2', 'gana', R`(가) $\lim_{x \to 0+} f(x) = \infty$
(나) 모든 실수 $x$에 대하여 $f(x) \ge 2$이다.`),
    b('q3', 'image', '<img src="/og-default.png" alt="" width="400" />', { imageWidth: 360, imageGray: true } as Partial<Block>),
    b('q4', 'choices', R`① $1$
② $\sqrt{3}$
③ $\dfrac{3}{8}$
④ 4
⑤ $-\dfrac{3\sqrt{10}}{10}$`),
  ],
  solution: [
    b('s1', 'case', R`$x>0$인 경우
$$
\begin{cases} x+1 & (x<1) \\ x^2 & (x \ge 1) \end{cases}
$$`),
    b('s2', 'svg', '/icons/phosphor/2.1.1/regular/acorn.svg', { svg_height: 120 } as Partial<Block>),
    b('s3', 'text', '<img src="/does-not-exist.png" width="200" />'),
  ],
};

declare global {
  interface Window { __h70: { pdf: string[][]; md: string[][]; ov: Record<string, string>[]; ready: boolean } }
}

export default function Dev70() {
  const [open, setOpen] = useState(true);
  useEffect(() => {
    window.__h70 = { pdf: [], md: [], ov: [], ready: true };
    // Overleaf: 실제 POST 대신 필드를 기록한다(외부 전송 금지)
    HTMLFormElement.prototype.submit = function submit(this: HTMLFormElement) {
      const rec: Record<string, string> = { action: this.action, target: this.target };
      for (const el of Array.from(this.elements) as HTMLInputElement[]) rec[el.name] = el.value;
      window.__h70.ov.push(rec);
    };
    window.open = (() => ({ close() {} })) as unknown as typeof window.open;
  }, []);
  return (
    <div style={{ padding: 20 }}>
      <button id="reopen" onClick={() => setOpen(true)}>열기</button>
      <ExportDialog
        open={open}
        onClose={() => setOpen(false)}
        title="표본: 미적분 #21"
        tabs={tabs}
        tabBlocks={tabBlocks}
        onPdf={async (ids) => { window.__h70.pdf.push(ids); setOpen(false); }}
        onMarkdown={(ids) => { window.__h70.md.push(ids); }}
      />
    </div>
  );
}
